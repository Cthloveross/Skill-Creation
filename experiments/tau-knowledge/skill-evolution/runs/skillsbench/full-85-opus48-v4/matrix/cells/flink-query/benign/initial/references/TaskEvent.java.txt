package clusterdata.datatypes;

import java.io.Serializable;

public class TaskEvent implements Serializable {
    public long timestamp;   // microseconds (field 0)
    public long jobId;       // field 2
    public int taskIndex;    // field 3
    public int eventType;    // field 5 (0 = SUBMIT)

    public TaskEvent() {}

    public TaskEvent(long timestamp, long jobId, int taskIndex, int eventType) {
        this.timestamp = timestamp;
        this.jobId = jobId;
        this.taskIndex = taskIndex;
        this.eventType = eventType;
    }

    // Returns null for malformed lines so the pipeline can filter them out.
    public static TaskEvent fromString(String line) {
        if (line == null || line.isEmpty()) return null;
        String[] f = line.split(",", -1);
        if (f.length < 6) return null;
        try {
            long ts = Long.parseLong(f[0].trim());
            long job = Long.parseLong(f[2].trim());
            int task = f[3].trim().isEmpty() ? -1 : Integer.parseInt(f[3].trim());
            int et = Integer.parseInt(f[5].trim());
            return new TaskEvent(ts, job, task, et);
        } catch (NumberFormatException e) {
            return null;
        }
    }

    @Override
    public String toString() {
        return "TaskEvent{ts=" + timestamp + ",job=" + jobId + ",task=" + taskIndex
                + ",type=" + eventType + "}";
    }
}
